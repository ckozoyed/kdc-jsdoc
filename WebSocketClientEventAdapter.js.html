/**
 * Convenience adapter for handling KDC WebSocket client events.
 *
 * Provides empty implementations of the WebSocket client
 * event callbacks. Applications may extend this class
 * and override only the methods they need.
 *
 * Event callbacks are invoked by WebSocketClient during
 * connection, transmission, and reception operations.
 *
 * @public
 */
export class WebSocketClientEventHandlerAdapter {

    /**
     * Called when the WebSocket connection is established.
     *
     * @param {WebSocketClient} client
     *     The connected WebSocket client.
     * @returns {void}
     */
    onConnectionEstablished(client) {}

    /**
     * Called when the WebSocket connection is closed.
     *
     * @param {WebSocketClient} client
     *     The disconnected WebSocket client.
     * @returns {void}
     */
    onConnectionClosed(client) {}

    /**
     * Called when an Agent is received from the server.
     *
     * @param {Agent} agent
     *     The received Agent.
     * @returns {void}
     */
    onReceivedAgent(agent) {}

    /**
     * Called before a synchronous Agent is transmitted.
     *
     * @param {Agent} agent
     *     The Agent being sent.
     * @returns {void}
     */
    onSendingSyncAgent(agent) {}

    /**
     * Called before an asynchronous Agent is transmitted.
     *
     * @param {Agent} agent
     *     The Agent being sent.
     * @returns {void}
     */
    onSendingAsyncAgent(agent) {}

    /**
     * Called when a Reply is received from the server.
     *
     * This callback is invoked for received Replies,
     * including those associated with synchronous requests.
     *
     * @param {Reply} reply
     *     The received Reply.
     * @returns {void}
     */
    onReceivedReply(reply) {}
}

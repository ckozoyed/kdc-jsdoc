import { Parameters } from "./Parameters.js";

/**
 * Configuration parameters shared by KDC-JS clients.
 *
 * Extends {@link Parameters} to provide the common
 * configuration inherited by client-specific parameter
 * classes.
 *
 * ClientParameters is intended to be extended by
 * transport-specific client parameter classes rather
 * than instantiated directly by application developers.
 *
 * @extends Parameters
 */
export class ClientParameters extends Parameters {

    /**
     * Creates client configuration parameters from
     * the supplied builder.
     *
     * Inherited configuration values are initialized
     * by the Parameters constructor.
     *
     * @param {ClientParametersBuilder} builder
     *     Builder containing the client configuration.
     */
    constructor(builder) {

        super(builder);

        if (new .target === ClientParameters) {
            Object.freeze(this);
        }

    }
}

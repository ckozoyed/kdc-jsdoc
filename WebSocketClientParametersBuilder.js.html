import { WebSocketClientParameters } from "./WebSocketClientParameters.js";
import { ClientParametersBuilder } from "./ClientParametersBuilder.js";

/**
 * Builder for configuring KDC-JS WebSocket clients.
 *
 * Extends ClientParametersBuilder with WebSocket-specific
 * configuration, including the server URL, synchronous
 * Reply timeout, and event handlers.
 *
 * Inherits common configuration methods for compression,
 * encryption, CRC checking, receive buffer size, and
 * Agent class registration.
 *
 * Default WebSocket settings:
 * - serverUrl: null (must be configured)
 * - syncReplyTimeout: 7500 milliseconds
 * - eventHandlers: empty array
 *
 * The builder is mutable and supports method chaining.
 * Calling build() validates the required configuration
 * and creates a WebSocketClientParameters instance.
 *
 * @public
 * @extends ClientParametersBuilder
 */
export class WebSocketClientParametersBuilder
        extends ClientParametersBuilder {

    /**
     * Creates a builder with default WebSocket settings.
     *
     * Initializes the inherited client configuration
     * and WebSocket-specific properties.
     */
    constructor() {
        super();

        /**
         * URL of the WebSocket server.
         *
         * Required before building the configuration.
         *
         * @type {string|null}
         */
        this.serverUrl = null;

        /**
         * Maximum time to wait for a synchronous Reply,
         * in milliseconds.
         *
         * Default: 7500 milliseconds.
         *
         * @type {number}
         */
        this.syncReplyTimeout = 7500;

        /**
         * Event handlers registered with the WebSocket client.
         *
         * @type {Array<Object>}
         */
        this.eventHandlers = [];

    }

    // ------------------------------------------------------------------
    // WebSocket-specific configuration
    // ------------------------------------------------------------------

    /**
     * Sets the WebSocket server URL.
     *
     * The URL normally uses the ws:// or wss:// scheme.
     *
     * @param {string} value
     *     WebSocket server URL.
     *
     * @returns {this}
     *     This builder for method chaining.
     *
     * @throws {Error}
     *     If value is null or undefined.
     */
    withServerUrl(value) {
        if (value === null || value === undefined) {
            throw new Error("serverUrl is required");
        }
        this.serverUrl = value;
        return this;
    }

    /**
     * Sets the synchronous Reply timeout.
     *
     * Determines how long sendSync() waits for a
     * matching Reply before rejecting the operation.
     *
     * @param {number} value
     *     Timeout in milliseconds. Must be a positive integer.
     *
     * @returns {this}
     *     This builder for method chaining.
     *
     * @throws {Error}
     *     If value is not a positive integer.
     */
    withSyncReplyTimeout(value) {

        if (!Number.isInteger(value) || value <= 0) {
            throw new Error(
                    "WebSocketClientParametersBuilder.withSyncReplyTimeout: " +
                    "value must be a positive integer"
                    );
        }

        this.syncReplyTimeout = value;

        return this;
    }

    // ------------------------------------------------------------------
    // Add Event Handler
    // ------------------------------------------------------------------

    /**
     * Registers an event handler with the WebSocket client.
     *
     * Multiple handlers may be registered. Each handler
     * is notified of applicable connection, Agent,
     * and Reply events.
     *
     * @param {Object} eventHandler
     *     Event handler implementing one or more
     *     WebSocket client callbacks.
     *
     * @returns {this}
     *     This builder for method chaining.
     *
     * @throws {Error}
     *     If eventHandler is null or otherwise falsy.
     */
    addEventHandler(eventHandler) {

        if (!eventHandler) {
            throw new Error("ParametersBuilder.addEventHandler: invalid event handler");
        }

        this.eventHandlers.push(eventHandler);

        return this;

    }

    // ------------------------------------------------------------------
    // Build
    // ------------------------------------------------------------------

    /**
     * Validates the required WebSocket configuration.
     *
     * Currently verifies that a server URL has been supplied.
     *
     * @returns {void}
     *
     * @throws {Error}
     *     If serverUrl is null or undefined.
     *
     * @private
     */
    validate() {
        if (this.serverUrl === null || this.serverUrl === undefined) {
            throw new Error("WebSocketClientParametersBuilder: serverUrl must be set");
        }
    }

    /**
     * Validates the configuration and creates
     * WebSocketClientParameters.
     *
     * @returns {WebSocketClientParameters}
     *     Configured WebSocket client parameters.
     *
     * @throws {Error}
     *     If the required server URL has not been configured.
     */
    build() {
        this.validate();
        return new WebSocketClientParameters(this);
    }
}

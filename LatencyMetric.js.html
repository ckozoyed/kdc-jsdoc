/**
 * Defines the latency metrics used by KDC to identify
 * stages of Agent and Reply processing.
 *
 * These constants provide consistent metric names
 * across KDC-Java and KDC-JS.
 *
 * Metrics describe the following operations:
 *
 * Agent processing:
 * - AGENT_SERIALIZE: Agent serialization
 * - AGENT_SEND: Agent transmission
 * - AGENT_RECEIVE: Agent reception
 * - AGENT_DESERIALIZE: Agent deserialization
 * - AGENT_QUEUE: Time spent waiting in the Agent queue
 * - AGENT_VISIT: Agent execution through its API
 *
 * Reply processing:
 * - REPLY_WAIT: Time spent waiting for a Reply
 * - REPLY_RECEIVE: Reply reception
 * - REPLY_DESERIALIZE: Reply deserialization
 * - REPLY_SERIALIZE: Reply serialization
 * - REPLY_SEND: Reply transmission
 *
 * The object is frozen to prevent modification of
 * metric definitions at runtime.
 *
 * These constants identify metric types; their presence
 * does not imply that every metric is currently measured
 * by KDC-JS.
 *
 * @public
 * @readonly
 * @enum {string}
 */
export const LatencyMetric = Object.freeze({
    AGENT_SERIALIZE: "AGENT_SERIALIZE",
    AGENT_SEND: "AGENT_SEND",
    AGENT_RECEIVE: "AGENT_RECEIVE",
    AGENT_DESERIALIZE: "AGENT_DESERIALIZE",
    AGENT_QUEUE: "AGENT_QUEUE",
    AGENT_VISIT: "AGENT_VISIT",
    REPLY_WAIT: "REPLY_WAIT",
    REPLY_RECEIVE: "REPLY_RECEIVE",
    REPLY_DESERIALIZE: "REPLY_DESERIALIZE",
    REPLY_SERIALIZE: "REPLY_SERIALIZE",
    REPLY_SEND: "REPLY_SEND"
});

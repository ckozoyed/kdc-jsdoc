import { AgentPacketV1 } from "./AgentPacketV1.js";

/**
 * Factory for creating version-specific KDC Agent packets.
 *
 * Provides a centralized mechanism for selecting the
 * appropriate packet implementation based on the KDC
 * wire protocol version.
 *
 * Used internally by KDC serialization and
 * deserialization components.
 *
 * This class is an internal implementation detail and
 * is not part of the public developer API.
 *
 * @private
 */
export class AgentPacket {

    /**
     * Wire protocol version 1.
     *
     * @type {number}
     * @readonly
     */
    static VERSION_1 = 1;

    /**
     * Creates an Agent packet for the specified
     * wire protocol version.
     *
     * Defaults to version 1.
     *
     * @param {number} [version=AgentPacket.VERSION_1]
     *     Requested packet protocol version.
     *
     * @returns {AgentPacketV1}
     *     Packet implementation for the requested version.
     *
     * @throws {Error}
     *     If the requested packet version is unsupported.
     */
    static create(version = AgentPacket.VERSION_1) {

        switch (version) {

            case AgentPacket.VERSION_1:
                return new AgentPacketV1();

            default:
                throw new Error(`Unsupported packet version ${version}`);
        }
    }
}

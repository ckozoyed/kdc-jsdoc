import { AgentSerializer } from "./AgentSerializer.js";

/**
 * Base immutable configuration for all KDC-JS clients and servers.
 *
 * This models the same protocol-level concerns as the Java Parameters class.
 */
export class Parameters {

    constructor(builder) {

        if (!builder) {
            throw new Error("Parameters: builder is required");
        }

        this.autoStart = builder.autoStart;
        this.maxReceiveBufferSize = builder.maxReceiveBufferSize;
        this.compressed = builder.compressed;
        this.encryptionKey = builder.encryptionKey;
        this.crcCheck = builder.crcCheck;
        this.agentRegistry = builder.agentRegistry;
        this.agentSerializer = new AgentSerializer(this);
        this.traceSerialization = true;

        if (new .target === Parameters) {
            Object.freeze(this);
        }

    }
}

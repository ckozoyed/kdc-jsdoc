/**
 * Records latency measurement timestamps for a KDC Agent.
 *
 * A LatencyRecord maintains start and stop timelines for
 * the latency metrics defined by its associated context.
 *
 * Each metric occupies a corresponding position in both
 * timelines. The context determines the available metrics
 * and their ordering.
 *
 * The startTimeLine, stopTimeLine, and latencyContext
 * properties form part of the serialized KDC wire contract.
 *
 * The runtime context reference is maintained separately
 * and is excluded from normal property enumeration.
 *
 * @class
 */
export class LatencyRecord {

    /**
     * Creates a latency record for the specified context.
     *
     * Both timelines are initialized to zero for every
     * metric defined by the context.
     *
     * @param {*} latencyContext
     *     Context defining the available latency metrics.
     *     Must provide size(), indexOf(), and name.
     *
     * @throws {TypeError}
     *     If latencyContext is null.
     */
    constructor(latencyContext) {

        if (latencyContext === null) {
            throw new TypeError(
                    "latencyContext must not be null"
                    );
        }

        /**
         * Start timestamps for each latency metric.
         *
         * Array positions correspond to the metric ordering
         * defined by the latency context.
         *
         * @type {number[]}
         */
        this.startTimeLine =
                new Array(latencyContext.size()).fill(0);

        /**
         * Stop timestamps for each latency metric.
         *
         * Array positions correspond to the metric ordering
         * defined by the latency context.
         *
         * @type {number[]}
         */
        this.stopTimeLine =
                new Array(latencyContext.size()).fill(0);

        /**
         * Name of the latency context associated with
         * this record.
         *
         * @type {string}
         */
        this.latencyContext = latencyContext.name;

        /**
         * Runtime reference to the latency context.
         *
         * This property is non-enumerable and is not
         * included in normal object serialization.
         *
         * @private
         */
        Object.defineProperty(this, "_context", {
            value: latencyContext,
            writable: false,
            enumerable: false
        });
    }

    /**
     * Marks the specified latency metric as not applicable.
     *
     * Both start and stop timeline entries are assigned
     * the sentinel value 1, distinguishing an inapplicable
     * metric from an uninitialized metric (0).
     *
     * This method is intended for internal KDC runtime use
     * and is not part of the public developer API.
     *
     * @private
     *
     * @param {*} metric
     *     Latency metric to mark as not applicable.
     *
     * @returns {void}
     */
    metricNotApplicable(metric) {

        const index =
                this._context.indexOf(metric);

        this.startTimeLine[index] = 1;
        this.stopTimeLine[index] = 1;
    }
}

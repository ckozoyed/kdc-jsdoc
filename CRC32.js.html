/**
 * Calculates a CRC-32 checksum using the same algorithm as
 * Java's java.util.zip.CRC32.
 *
 * Uses the standard reflected CRC-32 polynomial (0xEDB88320)
 * and a precomputed lookup table.
 *
 * Used internally by KDC to calculate and validate packet
 * checksums.
 *
 * This class is an internal implementation detail and is
 * not part of the public developer API.
 *
 * @private
 */
export class CRC32 {

    /**
     * Precomputed CRC-32 lookup table.
     *
     * @private
     * @type {Uint32Array}
     */
    static TABLE = CRC32.buildTable();

    /**
     * Creates a CRC-32 accumulator initialized to the
     * standard CRC-32 starting value.
     */
    constructor() {
        this.crc = 0xFFFFFFFF;
    }

    /**
     * Updates the running CRC-32 checksum with the
     * supplied bytes.
     *
     * Multiple calls accumulate into the same checksum.
     *
     * @param {Uint8Array} bytes
     *     Bytes to include in the checksum.
     *
     * @returns {void}
     *
     * @throws {TypeError}
     *     If bytes is not a Uint8Array.
     */
    update(bytes) {

        if (!(bytes instanceof Uint8Array)) {
            throw new TypeError("CRC32.update expects Uint8Array");
        }

        let crc = this.crc;

        for (const b of bytes) {
            crc =
                    (crc >>> 8)
                    ^
                    CRC32.TABLE[(crc ^ b) & 0xFF];
        }

        this.crc = crc;
    }

    /**
     * Returns the current CRC-32 checksum as an unsigned
     * 32-bit integer.
     *
     * Applies the standard final XOR operation.
     *
     * @returns {number}
     *     Unsigned CRC-32 checksum.
     */
    getValue() {
        return (~this.crc) >>> 0;
    }

    /**
     * Computes a CRC-32 checksum for the supplied bytes
     * using a new accumulator.
     *
     * @param {Uint8Array} bytes
     *     Bytes to include in the checksum.
     *
     * @returns {number}
     *     Unsigned CRC-32 checksum.
     *
     * @throws {TypeError}
     *     If bytes is not a Uint8Array.
     */
    static compute(bytes) {

        const crc = new CRC32();
        crc.update(bytes);

        return crc.getValue();

    }

    /**
     * Builds the lookup table used by the CRC-32 algorithm.
     *
     * Uses the reflected polynomial 0xEDB88320.
     *
     * @private
     * @returns {Uint32Array}
     *     Lookup table containing 256 CRC-32 entries.
     */
    static buildTable() {

        const table = new Uint32Array(256);

        for (let i = 0; i < 256; i++) {

            let c = i;

            for (let j = 0; j < 8; j++) {

                if ((c & 1) !== 0) {
                    c = 0xEDB88320 ^ (c >>> 1);
                } else {
                    c >>>= 1;
                }

            }

            table[i] = c >>> 0;

        }

        return table;

    }

}

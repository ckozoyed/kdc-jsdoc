import { ClientParameters } from "./ClientParameters.js";

/**
 * Configuration parameters for KDC-JS WebSocket clients.
 *
 * Extends ClientParameters with WebSocket-specific settings,
 * including the server URL, synchronous Reply timeout,
 * and registered event handlers.
 *
 * Instances are created using WebSocketClientParametersBuilder.
 *
 * The configuration object is frozen after construction
 * to prevent reassignment of its properties.
 *
 * Note that Object.freeze() is shallow; referenced objects
 * and arrays are not themselves frozen.
 *
 * @public
 * @extends ClientParameters
 */
export class WebSocketClientParameters extends ClientParameters {

    /**
     * Creates WebSocket client parameters from a builder.
     *
     * @param {WebSocketClientParametersBuilder} builder
     *     Builder containing the WebSocket client configuration.
     */
    constructor(builder) {

        super(builder);

        /**
         * URL of the WebSocket server.
         *
         * Typically uses the ws:// or wss:// scheme.
         *
         * @type {string}
         */
        this.serverUrl = builder.serverUrl;

        /**
         * Maximum time to wait for a synchronous Reply,
         * in milliseconds.
         *
         * @type {number}
         */
        this.syncReplyTimeout = builder.syncReplyTimeout;

        /**
         * Registered event handlers notified of WebSocket
         * connection, Agent, and Reply events.
         *
         * @type {Array<WebSocketClientEventHandlerAdapter>}
         */
        this.eventHandlers = builder.eventHandlers;

        Object.freeze(this);
    }
}

import { Agent } from "./Agent.js";

/**
 * Represents a response to an Agent operation in KDC-JS.
 *
 * Extends Agent to provide information about the outcome
 * of a request, including success status, an optional
 * response payload, and an optional error message.
 *
 * Reply objects participate in the same serialization
 * and class registration mechanisms as other Agents.
 *
 * The standard Reply class is automatically registered
 * by ParametersBuilder for interoperability with KDC-Java.
 *
 * @public
 * @extends Agent
 */
export class Reply extends Agent {

    /**
     * Creates a Reply with a successful default status
     * and no payload or error message.
     */
    constructor() {
        super();

        /**
         * Indicates whether the requested operation succeeded.
         *
         * Defaults to true.
         *
         * @type {boolean}
         */
        this.success = true;

        /**
         * Optional response payload.
         *
         * Defaults to null.
         *
         * @type {*}
         */
        this.payload = null;

        /**
         * Optional error message describing a failed operation.
         *
         * Defaults to null.
         *
         * @type {string|null}
         */
        this.errorMessage = null;
    }
}

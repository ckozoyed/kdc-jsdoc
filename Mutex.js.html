/**
 * Provides an asynchronous mutual exclusion lock.
 *
 * Ensures that asynchronous operations can acquire
 * exclusive access to a shared resource.
 *
 * When the lock is already held, additional callers
 * wait until the lock becomes available.
 *
 * Waiting callers are resumed in first-in, first-out
 * (FIFO) order.
 *
 * This class is an internal implementation detail
 * and is not part of the public developer API.
 *
 * @private
 */
export class Mutex {

    /**
     * Creates an initially unlocked mutex with
     * an empty queue of waiting operations.
     */
    constructor() {

        /**
         * Indicates whether the mutex is currently held.
         *
         * @private
         * @type {boolean}
         */
        this.locked = false;

        /**
         * Queue of Promise resolution callbacks for
         * operations waiting to acquire the lock.
         *
         * @private
         * @type {Array<Function>}
         */
        this.waiters = [];
    }

    /**
     * Acquires the mutex asynchronously.
     *
     * If the mutex is available, the caller acquires
     * it immediately.
     *
     * Otherwise, the caller waits until the current
     * holder releases the lock.
     *
     * The caller is responsible for invoking unlock()
     * after completing its protected operation.
     *
     * @returns {Promise<void>}
     *     Resolves when the caller has acquired the lock.
     */
    async lock() {

        if (!this.locked) {
            this.locked = true;
            return;
        }

        await new Promise(resolve => {
            this.waiters.push(resolve);
        });
    }

    /**
     * Releases the mutex.
     *
     * If operations are waiting, ownership is transferred
     * to the next waiting operation.
     *
     * Otherwise, the mutex becomes unlocked.
     *
     * @throws {Error}
     *     If the mutex is not currently locked.
     *
     * @returns {void}
     */
    unlock() {

        if (!this.locked) {
            throw new Error("Mutex.unlock: mutex is not locked");
        }

        const next = this.waiters.shift();

        if (next) {
            next();
        } else {
            this.locked = false;
        }
    }
}

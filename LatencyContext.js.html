import { LatencyMetric } from "./LatencyMetric.js";

/**
 * Represents a named collection of latency metrics
 * associated with a particular KDC processing context.
 *
 * Each context defines the metrics available for measurement
 * and their ordering within a LatencyRecord.
 *
 * Context instances are immutable.
 *
 * @private
 */
class LatencyContextValue {

    /**
     * Creates a latency context.
     *
     * @param {string} name
     *     Unique name identifying the context.
     * @param {string[]} metrics
     *     Ordered collection of latency metrics.
     */
    constructor(name, metrics) {

        /**
         * Name of the latency context.
         *
         * @type {string}
         * @readonly
         */
        this.name = name;

        /**
         * Ordered latency metrics associated with
         * this context.
         *
         * @type {ReadonlyArray<string>}
         * @readonly
         */
        this.metrics = Object.freeze([...metrics]);

        Object.freeze(this);
    }

    /**
     * Returns the number of metrics in this context.
     *
     * @returns {number}
     *     Number of latency metrics.
     */
    size() {
        return this.metrics.length;
    }

    /**
     * Returns the index of a latency metric within
     * this context.
     *
     * The index corresponds to the metric's position
     * in a LatencyRecord timeline.
     *
     * @param {string} metric
     *     Latency metric to locate.
     *
     * @returns {number}
     *     Zero-based index of the metric.
     *
     * @throws {RangeError}
     *     If the metric does not belong to this context.
     */
    indexOf(metric) {

        const index = this.metrics.indexOf(metric);

        if (index < 0) {
            throw new RangeError(
                `Metric ${metric} does not belong to context ${this.name}`
            );
        }

        return index;
    }

    /**
     * Returns the ordered latency metrics associated
     * with this context.
     *
     * The returned array is immutable.
     *
     * @returns {ReadonlyArray<string>}
     *     Ordered latency metrics.
     */
    getMetrics() {
        return this.metrics;
    }

    /**
     * Returns the name of this latency context.
     *
     * @returns {string}
     *     Context name.
     */
    toString() {
        return this.name;
    }
}

/**
 * Defines the supported KDC latency measurement contexts.
 *
 * Each context specifies the latency metrics applicable
 * to a particular stage of Agent communication.
 *
 * The ordering of metrics determines their positions
 * in LatencyRecord start and stop timelines.
 *
 * Context names and metric ordering must remain consistent
 * with KDC-Java for cross-language interoperability.
 *
 * @readonly
 * @enum {LatencyContextValue}
 */
export const LatencyContext = Object.freeze({

    /**
     * Asynchronous Agent sending context.
     *
     * Measures Agent serialization and transmission.
     */
    SEND_ASYNC: new LatencyContextValue(
        "SEND_ASYNC",
        [
            LatencyMetric.AGENT_SERIALIZE,
            LatencyMetric.AGENT_SEND
        ]
    ),

    /**
     * Synchronous Agent sending context.
     *
     * Includes sending-side measurements, receiving-side
     * Agent processing measurements returned with the Reply,
     * and synchronous reply-wait/receive measurements.
     */
    SEND_SYNC: new LatencyContextValue(
        "SEND_SYNC",
        [
            LatencyMetric.AGENT_SERIALIZE,
            LatencyMetric.AGENT_SEND,

            // Completed on receiver, returned with Reply
            LatencyMetric.AGENT_RECEIVE,
            LatencyMetric.AGENT_DESERIALIZE,
            LatencyMetric.AGENT_QUEUE,
            LatencyMetric.AGENT_VISIT,

            // Local synchronous wait/reply processing
            LatencyMetric.REPLY_WAIT,
            LatencyMetric.REPLY_RECEIVE,
            LatencyMetric.REPLY_DESERIALIZE
        ]
    ),

    /**
     * Asynchronous Agent receiving context.
     *
     * Measures Agent reception, deserialization,
     * queue processing, and visitor execution.
     */
    RECEIVE_ASYNC: new LatencyContextValue(
        "RECEIVE_ASYNC",
        [
            LatencyMetric.AGENT_RECEIVE,
            LatencyMetric.AGENT_DESERIALIZE,
            LatencyMetric.AGENT_QUEUE,
            LatencyMetric.AGENT_VISIT
        ]
    ),

    /**
     * Synchronous Agent receiving context.
     *
     * Measures Agent reception and processing,
     * followed by Reply serialization and transmission.
     */
    RECEIVE_SYNC: new LatencyContextValue(
        "RECEIVE_SYNC",
        [
            LatencyMetric.AGENT_RECEIVE,
            LatencyMetric.AGENT_DESERIALIZE,
            LatencyMetric.AGENT_QUEUE,
            LatencyMetric.AGENT_VISIT,
            LatencyMetric.REPLY_SERIALIZE,
            LatencyMetric.REPLY_SEND
        ]
    )
});

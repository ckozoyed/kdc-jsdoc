import { AgentClassRegistry } from "./AgentClassRegistry.js";
import { Parameters } from "./Parameters.js";
import { Agent } from "./Agent.js";
import { Reply } from "./Reply.js";

/**
 * Builder for configuring KDC-JS communication parameters.
 *
 * Provides default configuration values and fluent methods
 * for customizing serialization, compression, encryption,
 * CRC checking, and Agent class registration.
 *
 * Standard Agent and Reply classes are automatically
 * registered for interoperability with KDC-Java.
 *
 * The builder is mutable. Calling build() creates a
 * Parameters instance using the current configuration.
 *
 * Default configuration:
 * - autoStart: true
 * - maxReceiveBufferSize: 1,048,576 bytes (1 MiB)
 * - compressed: false
 * - encryptionKey: null
 * - crcCheck: false
 * - agentRegistry: new AgentClassRegistry
 *
 * @public
 */
export class ParametersBuilder {

    /**
     * Creates a builder with default configuration values
     * and registers the standard KDC Agent classes.
     */
    constructor() {

        /** @type {boolean} */
        this.autoStart = true;

        /** @type {number} */
        this.maxReceiveBufferSize = 1024 * 1024;

        /** @type {boolean} */
        this.compressed = false;

        /** @type {Uint8Array|null} */
        this.encryptionKey = null;

        /** @type {boolean} */
        this.crcCheck = false;

        /** @type {AgentClassRegistry} */
        this.agentRegistry = new AgentClassRegistry();

        this.registerStandardAgents();
    }

    /**
     * Registers the standard KDC Agent and Reply classes
     * using their corresponding Java class names.
     *
     * @private
     * @returns {void}
     */
    registerStandardAgents() {
        this.agentRegistry.register(
                Agent,
                "com.kristensoft.kdc.Agent"
                );

        this.agentRegistry.register(
                Reply,
                "com.kristensoft.kdc.Reply"
                );
    }

    // ------------------------
    // Scalar "with" methods
    // ------------------------

    // Not currently enabled:
    //
    // withAutoStart(value) {
    //     this.autoStart = Boolean(value);
    //     return this;
    // }

    /**
     * Sets the maximum permitted receive buffer size.
     *
     * @param {number} value
     *     Maximum receive buffer size in bytes.
     *
     * @returns {this}
     *     This builder for method chaining.
     *
     * @throws {Error}
     *     If value is not a positive integer.
     */
    withMaxReceiveBufferSize(value) {
        if (!Number.isInteger(value) || value <= 0) {
            throw new Error("maxReceiveBufferSize must be a positive integer");
        }
        this.maxReceiveBufferSize = value;
        return this;
    }

    /**
     * Enables or disables CRC checking for Agent packets.
     *
     * @param {boolean} value
     *     Whether CRC checking is enabled.
     *
     * @returns {this}
     *     This builder for method chaining.
     */
    withCrcCheck(value) {
        this.crcCheck = Boolean(value);
        return this;
    }

    /**
     * Enables or disables GZIP compression.
     *
     * @param {boolean} value
     *     Whether compression is enabled.
     *
     * @returns {this}
     *     This builder for method chaining.
     */
    withCompressed(value) {
        this.compressed = Boolean(value);
        return this;
    }

    /**
     * Configures the encryption key used for AES-GCM.
     *
     * A null value disables encryption.
     *
     * @param {Uint8Array|null} value
     *     Raw AES encryption key, or null.
     *
     * @returns {this}
     *     This builder for method chaining.
     */
    withEncryptionKey(value) {
        this.encryptionKey = value;
        return this;
    }

    // ------------------------
    // Registry-style methods
    // ------------------------

    // Not for KDC-JS 1.0:
    //
    // registerApi(visitorMethodName, apiInstance) {
    //     this.apiRegistry.register(visitorMethodName, apiInstance);
    //     return this;
    // }
    //
    // registerApis(apiMap) {
    //     if (!(apiMap instanceof Map)) {
    //         throw new Error("registerApis requires a Map");
    //     }
    //     for (const [k, v] of apiMap.entries()) {
    //         this.registerApi(k, v);
    //     }
    //     return this;

    /**
     * Registers a JavaScript Agent class with its
     * corresponding Java fully qualified class name.
     *
     * Used to reconstruct the correct JavaScript class
     * when receiving serialized Agents from KDC-Java.
     *
     * @param {Function} jsClass
     *     JavaScript Agent constructor.
     *
     * @param {string} javaFqcn
     *     Fully qualified Java class name.
     *
     * @returns {this}
     *     This builder for method chaining.
     */
    registerAgent(jsClass, javaFqcn) {
        this.agentRegistry.register(jsClass, javaFqcn);
        return this;
    }

    /**
     * Registers multiple Agent classes.
     *
     * Each Map entry associates a JavaScript constructor
     * with its corresponding Java fully qualified class name.
     *
     * @param {Map<Function, string>} agentMap
     *     Map of JavaScript constructors to Java class names.
     *
     * @returns {this}
     *     This builder for method chaining.
     *
     * @throws {Error}
     *     If agentMap is not a Map.
     */
    registerAgents(agentMap) {
        if (!(agentMap instanceof Map)) {
            throw new Error("registerAgents requires a Map");
        }
        for (const [k, v] of agentMap.entries()) {
            this.registerAgent(k, v);
        }
        return this;
    }

    // ------------------------
    // Build
    // ------------------------

    /**
     * Creates a Parameters instance from the current
     * builder configuration.
     *
     * @returns {Parameters}
     *     Configured Parameters instance.
     */
    build() {
        return new Parameters(this);
    }
}

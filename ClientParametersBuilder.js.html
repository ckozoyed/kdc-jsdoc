import { ClientParameters } from "./ClientParameters.js";
import { ParametersBuilder } from "./ParametersBuilder.js";

/**
 * Base builder for KDC-JS client configuration parameters.
 *
 * Extends {@link ParametersBuilder} to provide a common
 * foundation for transport-specific client builders.
 *
 * Maintains a configuration hierarchy consistent with
 * KDC-Java.
 *
 * @extends ParametersBuilder
 */
export class ClientParametersBuilder extends ParametersBuilder {

    /**
     * Creates a client parameters builder.
     */
    constructor() {
        super();
    }

    /**
     * Builds a ClientParameters instance using the
     * current builder configuration.
     *
     * @returns {ClientParameters}
     *     Configured client parameters.
     */
    build() {
        return new ClientParameters(this);
    }
}

import { KristenObjectsSerializer } from "./KristenObjectsSerializer.js";
import { AgentPacket } from "./AgentPacket.js";

/**
 * Serializes and deserializes KDC Agents using the configured
 * object serializer and versioned wire packet format.
 *
 * Responsibilities include:
 * - Resolving registered Agent class names.
 * - Serializing and deserializing Agent payloads.
 * - Creating and reading versioned Agent packets.
 * - Configuring packet CRC, compression, and encryption flags.
 * - Validating incoming packet configuration.
 * - Enforcing the maximum receive buffer size.
 *
 * Agent class registrations and serialization settings
 * are supplied through KDC Parameters.
 *
 * This class is an internal implementation detail and
 * is not part of the public developer API.
 *
 * @private
 */
export class AgentSerializer {

    /**
     * Creates an Agent serializer using the supplied parameters.
     *
     * @param {Object} parameters
     *     KDC serialization and Agent registration parameters.
     *
     * @throws {Error}
     *     If parameters are not supplied.
     */
    constructor(parameters) {

        if (!parameters) {
            throw new Error("AgentSerializer: parameters required");
        }

        this.objectSerializer = new KristenObjectsSerializer(parameters);
        this.crcCheck = parameters.crcCheck;
        this.compressed = parameters.compressed;
        this.encrypted = parameters.encryptionKey != null;
        this.agentRegistry = parameters.agentRegistry;
        this.maxReceiveBufferSize = parameters.maxReceiveBufferSize;
        Object.freeze(this);
    }

    /**
     * Serializes an Agent into a versioned KDC wire packet.
     *
     * Resolves the registered Java class name, serializes the
     * Agent payload, and applies the configured packet flags.
     *
     * @param {Object} agent
     *     Agent instance to serialize.
     *
     * @returns {Promise<Uint8Array>}
     *     Serialized wire packet.
     *
     * @throws {TypeError}
     *     If the Agent is null or undefined.
     *
     * @throws {Error}
     *     If serialization or class resolution fails.
     */
    async serialize(agent) {

        if (!agent) {
            throw new TypeError("AgentSerializer.serialize: agent is null");
        }

        const className = this.agentRegistry.resolve(agent);

        const payload = await this.objectSerializer.serialize(agent);

        const packet = AgentPacket.create();

        packet.className = className;
        packet.payload = payload;
        packet.setCrcCheck(this.crcCheck);
        packet.setCompressed(this.objectSerializer.compressed);
        packet.setEncrypted(this.objectSerializer.encryptionKey != null);

        return packet.writePacketToWire();

    }

    /**
     * Deserializes a KDC wire packet into a registered Agent.
     *
     * Validates the receive buffer size, packet format,
     * compression and encryption configuration, and Agent
     * class registration before reconstructing the Agent.
     *
     * @param {Uint8Array} bytes
     *     Serialized KDC wire packet.
     *
     * @returns {Promise<Object>}
     *     Deserialized Agent instance.
     *
     * @throws {Error}
     *     If the packet is invalid, exceeds the receive
     *     buffer limit, has incompatible configuration,
     *     or references an unregistered Agent class.
     */
    async deserialize(bytes) {

        if (!bytes) {
            throw new Error("Bytes not defined in deserialize.");
        }

        if (bytes.length > this.maxReceiveBufferSize) {
            throw new Error("Bytes length too big.  size=" + bytes.length + ", max=" + this.maxReceiveBufferSize);
        }

        const packet = AgentPacket.create();

        packet.readPacketFromWire(bytes);

        if (packet.compressed !== this.objectSerializer.compressed) {
            throw new Error(
                    `Compression configuration mismatch. ` +
                    `packet=${packet.compressed}, ` +
                    `serializer=${this.objectSerializer.compressed}`);
        }
        const serializerEncrypted = this.objectSerializer.encryptionKey != null;

        if (packet.encrypted !== serializerEncrypted) {
            throw new Error(
                    `Encryption configuration mismatch. ` +
                    `packet=${packet.encrypted}, ` +
                    `serializer=${serializerEncrypted}`);
        }

        if (!this.agentRegistry.has(packet.className)) {
            throw new Error(`Agent class not registered: ${packet.className}`);
        }

        const ctor = this.agentRegistry.get(packet.className);

        return await this.objectSerializer.deserialize(
                packet.payload,
                ctor);
    }
}
